local NOME = "AdminAbuseEAB"

local function limparAntigas()
    pcall(function()
        local a = CG:FindFirstChild(NOME)
        if a then a:Destroy() end
    end)
    pcall(function()
        local pg = P:FindFirstChild("PlayerGui")
        if pg and pg:FindFirstChild(NOME) then pg[NOME]:Destroy() end
    end)
    pcall(function()
        if gethui and gethui():FindFirstChild(NOME) then gethui()[NOME]:Destroy() end
    end)
    pcall(function()
        for _, gui in ipairs(CG:GetChildren()) do
            if gui.Name == NOME then gui:Destroy() end
        end
    end)
    pcall(function()
        local pg = P:FindFirstChild("PlayerGui")
        if pg then
            for _, gui in ipairs(pg:GetChildren()) do
                if gui.Name == NOME then gui:Destroy() end
            end
        end
    end)
end

local dragging, dragInput, dragStart, startPos = false, nil, nil, nil

titulo.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
        dragInput = input
    end
end)

titulo.InputChanged:Connect(function(input)
    if input == dragInput
    and (input.UserInputType == Enum.UserInputType.MouseMovement
      or input.UserInputType == Enum.UserInputType.Touch) then
        if not dragging then return end
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

titulo.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
        dragInput = nil
    end
end)

local function getHRP()
    local chr = P.Character
    if not chr then return nil end
    return chr:FindFirstChild("HumanoidRootPart")
end

local function getHum()
    local chr = P.Character
    if not chr then return nil end
    return chr:FindFirstChildOfClass("Humanoid")
end

local function pegarParte(p)
    local o = p.Parent
    while o and not o:IsA("BasePart") do
        if o == workspace then return nil end
        o = o.Parent
    end
    return o
end

-- função pra criar botão
local function criarBotao(texto, yPos, callback)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, -20, 0, 42)
    btn.Position = UDim2.new(0, 10, 0, yPos)
    btn.BackgroundColor3 = ROXO
    btn.BorderSizePixel = 0
    btn.Text = texto
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextSize = isMobile and 13 or 14
    btn.Font = Enum.Font.GothamBold
    btn.AutoButtonColor = false
    btn.Parent = main
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)

    local estado = false
    btn.MouseButton1Click:Connect(function()
        estado = not estado
        btn.BackgroundColor3 = estado and ROXO_ATIV or ROXO
        if callback then callback(estado) end
    end)

    if not isMobile then
        btn.MouseEnter:Connect(function()
            if not estado then
                TS:Create(btn, TweenInfo.new(0.1), { BackgroundColor3 = ROXO_HOV }):Play()
            end
        end)
        btn.MouseLeave:Connect(function()
            if not estado then
                TS:Create(btn, TweenInfo.new(0.1), { BackgroundColor3 = ROXO }):Play()
            end
        end)
    end

    return btn
end

local freezeOn = false
local freezeConn = nil
local wsSalvo = 16
local jpSalvo = 50

local buyOn = false
local buyProcessando = false

-- ================================================================
-- 1. FREEZE
-- ================================================================
local function aplicarFreeze()
    local hrp = getHRP()
    local hum = getHum()
    if not hrp then return end

    local posTravada = hrp.CFrame

    if hum then
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum.JumpHeight = 0
    end

    if freezeConn then freezeConn:Disconnect() end
    freezeConn = RS.RenderStepped:Connect(function()
        if hrp and hrp.Parent then
            hrp.Anchored = true
            hrp.CFrame = posTravada
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

local function removerFreeze()
    if freezeConn then freezeConn:Disconnect() freezeConn = nil end
    local hrp = getHRP()
    local hum = getHum()
    if hrp then hrp.Anchored = false end
    if hum then hum.WalkSpeed = wsSalvo hum.JumpPower = jpSalvo hum.JumpHeight = 7.2 end
end

criarBotao("❄ Freeze", 44, function(on)
    freezeOn = on
    if on then aplicarFreeze() else removerFreeze() end
end)

local function visivel(prompt)
    local pt = pegarParte(prompt)
    if not pt then return true end
    local camPos = C.CFrame.Position
    if (camPos - pt.Position).Magnitude > 50 then return false end
    local dir = (pt.Position - camPos).Unit
    local ang = math.deg(math.acos(math.clamp(dir:Dot(C.CFrame.LookVector), -1, 1)))
    if ang > 80 then return false end

    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = {P.Character, C}
    rp.IgnoreWater = true
    local r = workspace:Raycast(camPos, pt.Position - camPos, rp)
    if r and r.Instance ~= pt and not r.Instance:IsDescendantOf(pt) and not pt:IsDescendantOf(r.Instance) then
        return false
    end
    return true
end

local function executarBuy()
    if buyProcessando then return end
    buyProcessando = true

    local lista = {}
    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("ProximityPrompt") and o.Enabled and visivel(o) then
            table.insert(lista, o)
        end
    end

    if #lista == 0 then buyProcessando = false return end

    table.sort(lista, function(a, b)
        local pa, pb = pegarParte(a), pegarParte(b)
        if not pa then return false end
        if not pb then return true end
        return (C.CFrame.Position - pa.Position).Magnitude < (C.CFrame.Position - pb.Position).Magnitude
    end)

    for _, prompt in ipairs(lista) do
        if not buyOn then break end
        pcall(function()
            prompt:InputHoldBegin()
            task.wait(prompt.HoldDuration + 0.05)
            prompt:InputHoldEnd()
        end)
        task.wait(0.25)
    end

    buyProcessando = false
end

task.spawn(function()
    while sg.Parent do
        task.wait(0.5)
        if buyOn and not buyProcessando then executarBuy() end
    end
end)

criarBotao("🛒 Auto Buy", 90, function(on)
    buyOn = on
end)
