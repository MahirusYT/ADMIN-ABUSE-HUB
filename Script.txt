    end)
end

local dragging, dragInput, dragStart, startPos = false, nil, nil, nil

local function tornarArrastavel(handle)
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = main.Position
            dragInput = input
        end
    end)
    handle.InputChanged:Connect(function(input)
        if input == dragInput
        and (input.UserInputType == Enum.UserInputType.MouseMovement
          or input.UserInputType == Enum.UserInputType.Touch) then
            if not dragging then return end
            local delta = input.Position - dragStart
            main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
    handle.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            dragInput = nil
        end
    end)
end

-- arrasta SÓ pelo título (pra não bugar os botões)

-- ================================================================
-- HELPERS
-- ================================================================
local function getHRP()
    local chr = P.Character
    if not chr then return nil end
    return chr:FindFirstChild("HumanoidRootPart")
end

local function getHum()
    local chr = P.Character
    if not chr then return nil end
    return chr:FindFirstChildOfClass("Humanoid")
end

local function pegarParte(p)
    local o = p.Parent
    while o and not o:IsA("BasePart") do
        if o == workspace then return nil end
        o = o.Parent
    end
    return o
end

-- ================================================================
-- 1. FREEZE
-- ================================================================
local function aplicarFreeze()
    local hrp = getHRP()
    local hum = getHum()
    if not hrp then return end
    local posTravada = hrp.CFrame
    if hum then hum.WalkSpeed = 0 hum.JumpPower = 0 hum.JumpHeight = 0 end
    if freezeConn then freezeConn:Disconnect() end
    freezeConn = RS.RenderStepped:Connect(function()
        if hrp and hrp.Parent then
            hrp.Anchored = true
            hrp.CFrame = posTravada
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

local function removerFreeze()
    if freezeConn then freezeConn:Disconnect() freezeConn = nil end
    local hrp = getHRP()
    local hum = getHum()
    if hrp then hrp.Anchored = false end
    if hum then hum.WalkSpeed = walkspeedSalvo hum.JumpPower = jumppowerSalvo hum.JumpHeight = 7.2 end
end

addToggle("❄ Freeze", function(on)
    freezeOn = on
    if on then aplicarFreeze() else removerFreeze() end
end)

local function pararFly()
    if flyConn then flyConn:Disconnect() flyConn = nil end
    if flyBV then flyBV:Destroy() flyBV = nil end
    if flyBG then flyBG:Destroy() flyBG = nil end
end

addToggle("🕊 Fly", function(on)
    flyOn = on
    if not on then pararFly() return end
    local hrp = getHRP()
    if not hrp then return end

    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    flyBV.Velocity = Vector3.zero
    flyBV.Parent = hrp

    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    flyBG.P = 1000
    flyBG.D = 50
    flyBG.Parent = hrp

    flyConn = RS.RenderStepped:Connect(function()
        if not flyOn or not hrp or not hrp.Parent then return end
        local moveDir = Vector3.zero
        if UIS:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + C.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - C.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - C.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + C.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
        if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end
        if moveDir.Magnitude > 0 then flyBV.Velocity = moveDir.Unit * 80 else flyBV.Velocity = Vector3.zero end
        flyBG.CFrame = C.CFrame
    end)
end)

-- 8. AUTO BUY
-- ================================================================
local function visivel(prompt)
    local pt = pegarParte(prompt)
    if not pt then return true end
    local camPos = C.CFrame.Position
    if (camPos - pt.Position).Magnitude > 50 then return false end
    local dir = (pt.Position - camPos).Unit
    local ang = math.deg(math.acos(math.clamp(dir:Dot(C.CFrame.LookVector), -1, 1)))
    if ang > 80 then return false end
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = {P.Character, C}
    rp.IgnoreWater = true
    local r = workspace:Raycast(camPos, pt.Position - camPos, rp)
    if r and r.Instance ~= pt and not r.Instance:IsDescendantOf(pt) and not pt:IsDescendantOf(r.Instance) then
        return false
    end
    return true
end

local function executarBuy()
    if buyProcessando then return end
    buyProcessando = true
    local lista = {}
    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("ProximityPrompt") and o.Enabled and visivel(o) then
            table.insert(lista, o)
        end
    end
    if #lista == 0 then buyProcessando = false return end
    table.sort(lista, function(a, b)
        local pa, pb = pegarParte(a), pegarParte(b)
        if not pa then return false end
        if not pb then return true end
        return (C.CFrame.Position - pa.Position).Magnitude < (C.CFrame.Position - pb.Position).Magnitude
    end)
    for _, prompt in ipairs(lista) do
        if not buyOn then break end
        pcall(function()
            prompt:InputHoldBegin()
            task.wait(prompt.HoldDuration + 0.05)
            prompt:InputHoldEnd()
        end)
        task.wait(0.25)
    end
    buyProcessando = false
end

task.spawn(function()
    while sg.Parent do
        task.wait(0.5)
        if buyOn and not buyProcessando then executarBuy() end
    end
end)
